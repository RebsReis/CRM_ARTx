CREATE OR REPLACE FUNCTION public.registrar_historico_lead()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    IF OLD.fase IS DISTINCT FROM NEW.fase THEN

        INSERT INTO historico_leads (
            lead_id,
            fase_anterior,
            fase_nova,
            movimentado_em
        )
        VALUES (
            OLD.id,
            OLD.fase,
            NEW.fase,
            CURRENT_TIMESTAMP
        );

    END IF;

    NEW.atualizado_em = CURRENT_TIMESTAMP;

    RETURN NEW;
END;
$function$
